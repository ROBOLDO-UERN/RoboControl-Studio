# ARQUIVO: server.py (Servidor Piper Offline com CORS p/ Painel Roboldo 2.0)
# Salve este HTML como "file.html" ou "index.html" na mesma pasta do modelo .onnx
# ============================================================================
import io
import json
import wave
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from pathlib import Path

from piper import PiperVoice
from piper.config import SynthesisConfig

ROOT = Path(__file__).resolve().parent
MODEL_PATH = ROOT / "pt_BR-faber-medium.onnx"
VOICE = PiperVoice.load(str(MODEL_PATH))
VOICE_NOISE_SCALE = 0.667 * (0.8 + (1.2 - 1.0) * 0.35)
VOICE_NOISE_W_SCALE = 0.8 * (0.8 + (1.2 - 1.0) * 0.45)

class TTSHandler(BaseHTTPRequestHandler):
    def log_message(self, format, *args):
        return

    def send_cors_headers(self):
        self.send_header("Access-Control-Allow-Origin", "*")
        self.send_header("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
        self.send_header("Access-Control-Allow-Headers", "Content-Type")

    def do_OPTIONS(self):
        self.send_response(204)
        self.send_cors_headers()
        self.end_headers()

    def send_json(self, status, payload):
        body = json.dumps(payload, ensure_ascii=False).encode("utf-8")
        self.send_response(status)
        self.send_cors_headers()
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self):
        if self.path in {"/", "/file.html", "/index.html"}:
            target_file = ROOT / ("file.html" if (ROOT / "file.html").exists() else "index.html")
            html = target_file.read_bytes()
            self.send_response(200)
            self.send_cors_headers()
            self.send_header("Content-Type", "text/html; charset=utf-8")
            self.send_header("Content-Length", str(len(html)))
            self.end_headers()
            self.wfile.write(html)
            return
        self.send_json(404, {"error": "Página não encontrada."})

    def do_POST(self):
        if self.path != "/api/synthesize":
            self.send_json(404, {"error": "Rota não encontrada."})
            return

        content_length = int(self.headers.get("Content-Length", "0"))
        raw_body = self.rfile.read(content_length) if content_length > 0 else b"{}"
        try:
            payload = json.loads(raw_body.decode("utf-8"))
        except Exception:
            self.send_json(400, {"error": "JSON inválido."})
            return

        text = str(payload.get("text", "")).strip()
        if not text:
            self.send_json(400, {"error": "Informe um texto para sintetizar."})
            return

        rate = float(payload.get("rate", 1.05))
        volume = max(0.1, min(1.0, float(payload.get("volume", 1.0))))
        syn_config = SynthesisConfig(
            length_scale=1.0 / rate,
            noise_scale=VOICE_NOISE_SCALE,
            noise_w_scale=VOICE_NOISE_W_SCALE,
            volume=volume,
        )

        buffer = io.BytesIO()
        with wave.open(buffer, "wb") as wav_file:
            VOICE.synthesize_wav(text, wav_file, syn_config=syn_config)

        audio = buffer.getvalue()
        self.send_response(200)
        self.send_cors_headers()
        self.send_header("Content-Type", "audio/wav")
        self.send_header("Content-Length", str(len(audio)))
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        self.wfile.write(audio)

if __name__ == "__main__":
    server = ThreadingHTTPServer(("127.0.0.1", 3000), TTSHandler)
    print("Roboldo 2.0 (Piper + Dual USB Studio) em http://127.0.0.1:3000")
    server.serve_forever()

  • "n\n" -> MODE_NEUTRAL_IDLE  | "a\n" -> Angry | "c\n" -> Confused | "f\n" -> Fear
  • "h\n" -> Happy              | "s\n" -> Sad   | "u\n" -> Surprised
  • "1\n" -> TALK_SYLLABLES     | "2\n" -> TALK_ANIMATED | "3\n" -> TALK_FAST
  • "RAW:earR,earL,neck,speed\n" -> Controle direto dos 3 servos (Ex: "RAW:140,40,105,10\n")
